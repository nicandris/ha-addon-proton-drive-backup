#include <tunables/global>

# The Supervisor renames this profile to the installed slug when it loads it.
# Deliberately broad on files and network: the add-on reads /data, stages
# archives in /tmp, runs node and the proton-drive binary, and talks HTTPS to
# Proton and HTTP to the Supervisor. What it takes away is everything else:
# mount, ptrace of other processes, module loading and every capability not
# listed below (Docker grants a container root more than this by default).
profile proton_drive_backup flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  file,
  network,
  unix,
  signal,
  ptrace (read) peer=@{profile_name},

  # Docker's defaults that a root process writing its own files can need.
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,

  deny mount,
  deny umount,
  deny pivot_root,
}
